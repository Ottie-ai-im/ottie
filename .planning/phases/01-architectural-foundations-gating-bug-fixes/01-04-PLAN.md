session.ts:1685-1698 currently dispatches via a chain of `dispatchVoiceAndControlMessage(msg) ?? dispatchAgentLifecycleMessage(msg) ?? ... ?? dispatchMiscMessage(msg)`. The router replaces this with explicit registration:

```typescript
// session/router.ts
import type { SessionInboundMessage } from "../messages.js";

export type RouterHandler = (msg: SessionInboundMessage) => Promise<void> | void;

export class MessageRouter {
  private readonly handlers = new Map<string, RouterHandler>();

  register(kind: string, handler: RouterHandler): void {
    if (this.handlers.has(kind)) {
      throw new Error(`Duplicate handler registration for kind=${kind}`);
    }
    this.handlers.set(kind, handler);
  }

  async dispatch(msg: SessionInboundMessage): Promise<void> {
    const handler = this.handlers.get(msg.type);
    if (!handler) {
      // No registered handler — fall through to the legacy session.ts path is the caller's responsibility (Strangler-Fig).
      throw new RouterMissError(msg.type);
    }
    await handler(msg);
  }

  has(kind: string): boolean {
    return this.handlers.has(kind);
  }
}

export class RouterMissError extends Error {
  constructor(public readonly kind: string) {
    super(`No handler registered for kind=${kind}`);
  }
}
```

  async handleResponse(
    agentId: string,
    requestId: string,
    response: AgentPermissionResponse,

// session/inflight-counter.ts
export class InflightCounter {
  private current = 0;
  private peakValue = 0;

  increment(): void {
    this.current++;
    if (this.current > this.peakValue) {
      this.peakValue = this.current;
    }
  }

  decrement(): void {
    this.current--;
  }

  get value(): number {
    return this.current;
  }

  get peak(): number {
    return this.peakValue;
  }

  resetPeak(): void {
    this.peakValue = this.current;
  }
}
